C) Crear repositorio github 
  # Crea repo y lo clona en directorio actual
  gh repo create nombre-de-tu-repo --public --clone

D) Subir archivos a un nuevo repo 
  cd nombre-repo 
  git init 
  git add .
  git commit -m 'Subimos archivos desde el terminal'
  gh repo create nombre-de-tu-repo --public --source=. --remote=nombre_remoto --push  
  git push nombre-remoto master

  # OTROS COMANDOS DE GITHUB 

  # Tambien podemos crearlo y subir los archivos todo de una, pero me gusta hacerlo paso a paso 
  gh repo create nombre-de-tu-repo --public --source=. --remote=nombre_remoto --push  

  # Si tuvieramos un repositorio ya creado y quisieramos cambiar la url para usar SSH:
  git remote set-url master git@github.com:rnoguer22/nombre-de-tu-repo.git

  # Añadir un remoto nuevo a un repositorio 
  git remote add nombre-remoto git@github.com:rnoguer22/nombre-de-tu-repo.git
